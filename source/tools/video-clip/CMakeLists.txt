cmake_minimum_required(VERSION 3.24)
project(ModynVideo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
 add_compile_options(/utf-8)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
add_library(video_core core/video.cpp)
target_link_libraries(video_core PUBLIC Qt6::Core)
add_executable(ModynVideo integration/adapter.cpp)
target_include_directories(ModynVideo PRIVATE ../../shared/sdk/cpp)
target_link_libraries(ModynVideo PRIVATE video_core)
add_executable(ModynVideoCli standalone/main.cpp)
target_link_libraries(ModynVideoCli PRIVATE video_core)
