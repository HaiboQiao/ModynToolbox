cmake_minimum_required(VERSION 3.24)
project(ModynFileHash VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
add_library(hash_core core/hash_core.cpp)
target_include_directories(hash_core PUBLIC core)
target_link_libraries(hash_core PUBLIC Qt6::Core)
add_executable(ModynHashCli standalone/main.cpp)
target_link_libraries(ModynHashCli PRIVATE hash_core)
option(MODYN_WITH_ADAPTER "Build the toolbox protocol adapter" ON)
if(MODYN_WITH_ADAPTER)
    add_executable(ModynHash integration/adapter.cpp)
    target_include_directories(ModynHash PRIVATE ../../shared/sdk/cpp)
    target_link_libraries(ModynHash PRIVATE hash_core)
endif()
