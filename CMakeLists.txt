cmake_minimum_required(VERSION 3.24)
project(GraphiteToolbox VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Test)
enable_testing()
add_library(graphite_shell app/ui/workbench.cpp app/ui/workbench.h app/catalog/package_catalog.cpp app/platform/windows/window_theme.cpp)
target_link_libraries(graphite_shell PUBLIC Qt6::Widgets)
target_include_directories(graphite_shell PUBLIC app)
add_executable(GraphiteToolboxApp WIN32 app/main.cpp)
target_link_libraries(GraphiteToolboxApp PRIVATE graphite_shell)
set_target_properties(GraphiteToolboxApp PROPERTIES OUTPUT_NAME GraphiteToolbox.App)
add_executable(graphite_shell_tests app/tests/unit/test_workbench.cpp)
target_link_libraries(graphite_shell_tests PRIVATE graphite_shell Qt6::Test)
add_test(NAME workbench COMMAND graphite_shell_tests -o -,txt -o workbench-results.txt,txt)
add_executable(graphite_worker app/tests/fixtures/worker.cpp)
target_link_libraries(graphite_worker PRIVATE Qt6::Core)
foreach(target GraphiteToolboxApp graphite_shell_tests graphite_worker)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME worker_transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/app/tests/integration/test_transport.py" "$<TARGET_FILE:graphite_worker>" "${CMAKE_SOURCE_DIR}")
if(WIN32)
  target_link_libraries(graphite_shell PRIVATE dwmapi)
  target_compile_options(graphite_shell PRIVATE /utf-8)
endif()
add_executable(graphite_catalog_tests app/tests/unit/test_package_catalog.cpp app/catalog/package_catalog.cpp)
target_include_directories(graphite_catalog_tests PRIVATE app)
target_link_libraries(graphite_catalog_tests PRIVATE Qt6::Test)
set_target_properties(graphite_catalog_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME package_catalog COMMAND graphite_catalog_tests -o -,txt -o catalog-results.txt,txt)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/tests/tmp")
set_tests_properties(workbench package_catalog worker_transport PROPERTIES
  ENVIRONMENT "QT_QPA_PLATFORM=offscreen;TEMP=${CMAKE_BINARY_DIR}/tests/tmp;TMP=${CMAKE_BINARY_DIR}/tests/tmp")
set_tests_properties(workbench package_catalog worker_transport PROPERTIES TIMEOUT 45)
