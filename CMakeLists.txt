cmake_minimum_required(VERSION 3.24)
project(GraphiteToolbox VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Test)
enable_testing()
add_library(graphite_shell app/ui/workbench.cpp app/ui/workbench.h)
target_link_libraries(graphite_shell PUBLIC Qt6::Widgets)
target_include_directories(graphite_shell PUBLIC app)
add_executable(GraphiteToolboxApp WIN32 app/main.cpp)
target_link_libraries(GraphiteToolboxApp PRIVATE graphite_shell)
set_target_properties(GraphiteToolboxApp PROPERTIES OUTPUT_NAME GraphiteToolbox.App)
add_executable(graphite_shell_tests app/tests/unit/test_workbench.cpp)
target_link_libraries(graphite_shell_tests PRIVATE graphite_shell Qt6::Test)
add_test(NAME workbench COMMAND graphite_shell_tests)
add_executable(graphite_worker app/tests/fixtures/worker.cpp)
target_link_libraries(graphite_worker PRIVATE Qt6::Core)
foreach(target GraphiteToolboxApp graphite_shell_tests graphite_worker)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
